cmake_minimum_required(VERSION 3.24)
project(BrahmaEngine VERSION 0.1.0 LANGUAGES CXX)
option(BRAHMA_ENGINE_BUILD_TESTS "Build Brahma Engine tests" ON)
option(BRAHMA_ENGINE_ENABLE_OPENMP "Enable OpenMP parallel CPU kernels" ON)
add_library(brahma_engine_core src/gguf_reader.cpp src/tensor_validator.cpp src/quantization.cpp src/tensor_ops.cpp src/tensor_loader.cpp src/tensor_registry.cpp src/qwen2_schema.cpp src/qwen2_block.cpp src/qwen2_runtime.cpp src/qwen2_tokenizer.cpp)
target_include_directories(brahma_engine_core PUBLIC include)
target_compile_features(brahma_engine_core PUBLIC cxx_std_20)
if(BRAHMA_ENGINE_ENABLE_OPENMP)
  find_package(OpenMP QUIET)
endif()
if(BRAHMA_ENGINE_ENABLE_OPENMP AND OpenMP_CXX_FOUND)
  target_link_libraries(brahma_engine_core PUBLIC OpenMP::OpenMP_CXX)
  target_compile_definitions(brahma_engine_core PRIVATE BRAHMA_ENGINE_USE_OPENMP=1)
endif()
if(MSVC)
  target_compile_options(brahma_engine_core PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(brahma_engine_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(brahma-engine src/main.cpp)
target_link_libraries(brahma-engine PRIVATE brahma_engine_core)
if(BRAHMA_ENGINE_BUILD_TESTS)
  enable_testing()
  add_executable(brahma-engine-tests tests/gguf_reader_tests.cpp)
  target_link_libraries(brahma-engine-tests PRIVATE brahma_engine_core)
  add_test(NAME gguf_reader_tests COMMAND brahma-engine-tests)
  add_executable(brahma-tensor-ops-tests tests/tensor_ops_tests.cpp)
  target_link_libraries(brahma-tensor-ops-tests PRIVATE brahma_engine_core)
  add_test(NAME tensor_ops_tests COMMAND brahma-tensor-ops-tests)
  add_executable(brahma-tensor-loader-tests tests/tensor_loader_tests.cpp)
  target_link_libraries(brahma-tensor-loader-tests PRIVATE brahma_engine_core)
  add_test(NAME tensor_loader_tests COMMAND brahma-tensor-loader-tests)
  add_executable(brahma-qwen2-schema-tests tests/qwen2_schema_tests.cpp)
  target_link_libraries(brahma-qwen2-schema-tests PRIVATE brahma_engine_core)
  add_test(NAME qwen2_schema_tests COMMAND brahma-qwen2-schema-tests)
  add_executable(brahma-qwen2-block-tests tests/qwen2_block_tests.cpp)
  target_link_libraries(brahma-qwen2-block-tests PRIVATE brahma_engine_core)
  add_test(NAME qwen2_block_tests COMMAND brahma-qwen2-block-tests)
  add_executable(brahma-qwen2-tokenizer-tests tests/qwen2_tokenizer_tests.cpp)
  target_link_libraries(brahma-qwen2-tokenizer-tests PRIVATE brahma_engine_core)
  add_test(NAME qwen2_tokenizer_tests COMMAND brahma-qwen2-tokenizer-tests)
  add_executable(brahma-quantization-tests tests/quantization_tests.cpp)
  target_link_libraries(brahma-quantization-tests PRIVATE brahma_engine_core)
  add_test(NAME quantization_tests COMMAND brahma-quantization-tests)
endif()
