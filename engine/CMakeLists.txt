cmake_minimum_required(VERSION 3.24)
project(BrahmaEngine VERSION 0.1.0 LANGUAGES CXX)
option(BRAHMA_ENGINE_BUILD_TESTS "Build Brahma Engine tests" ON)
add_library(brahma_engine_core src/gguf_reader.cpp src/tensor_validator.cpp src/quantization.cpp src/tensor_ops.cpp src/tensor_loader.cpp)
target_include_directories(brahma_engine_core PUBLIC include)
target_compile_features(brahma_engine_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(brahma_engine_core PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(brahma_engine_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(brahma-engine src/main.cpp)
target_link_libraries(brahma-engine PRIVATE brahma_engine_core)
if(BRAHMA_ENGINE_BUILD_TESTS)
  enable_testing()
  add_executable(brahma-engine-tests tests/gguf_reader_tests.cpp)
  target_link_libraries(brahma-engine-tests PRIVATE brahma_engine_core)
  add_test(NAME gguf_reader_tests COMMAND brahma-engine-tests)
  add_executable(brahma-tensor-ops-tests tests/tensor_ops_tests.cpp)
  target_link_libraries(brahma-tensor-ops-tests PRIVATE brahma_engine_core)
  add_test(NAME tensor_ops_tests COMMAND brahma-tensor-ops-tests)
endif()
